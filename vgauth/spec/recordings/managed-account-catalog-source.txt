async function createProduct(
  name: string,
  priceCents: number,
  recurring: boolean,
): Promise<string> {
  const response = await fetch(`${polarTwinUrl}/v1/products/`, {
    method: 'POST',
    headers: polarHeaders,
    body: JSON.stringify({
      name,
      ...(recurring ? { recurring_interval: 'month' } : {}),
      prices: [{ amount_type: 'fixed', price_amount: priceCents, price_currency: 'usd' }],
    }),
  });
  const body = (await response.json().catch(() => undefined)) as { id?: unknown } | undefined;
  if (!response.ok || typeof body?.id !== 'string') {
    throw new Error(`Polar Twin product seed failed for ${name} (${response.status}).`);
  }
  return body.id;
}

interface LocalAccountProjection {
  user: { id: string };
  entitlements: { generation: boolean };
}

async function accountForToken(
  accountUrl: string,
  token: string,
): Promise<LocalAccountProjection | undefined> {
  const response = await fetch(`${accountUrl}/v1/account`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!response.ok) return undefined;
  const value = (await response.json()) as Partial<LocalAccountProjection>;
  return typeof value.user?.id === 'string' && typeof value.entitlements?.generation === 'boolean'
    ? (value as LocalAccountProjection)
    : undefined;
}

async function requestJson(request: IncomingMessage): Promise<Record<string, unknown>> {
  const chunks: Buffer[] = [];
  for await (const chunk of request) {
    chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk));
  }
  try {
    const value = JSON.parse(Buffer.concat(chunks).toString('utf8')) as unknown;
    return value && typeof value === 'object' && !Array.isArray(value)
      ? (value as Record<string, unknown>)
      : {};
  } catch {
    return {};
  }
}

function sendJson(response: ServerResponse, status: number, body: unknown): void {
  response.statusCode = status;
  response.setHeader('Content-Type', 'application/json');
  response.setHeader('Cache-Control', 'no-store');
  response.end(JSON.stringify(body));
}

async function startLocalInferenceAuthority(
  accountUrl: string,
  grants: Map<string, ManagedInferenceGrant>,
): Promise<{ url: string; close(): Promise<void> }> {
  const server = createServer(async (request, response) => {
    try {
      if (request.method !== 'POST' || request.url !== '/inference/grant') {
        sendJson(response, 404, { error: 'Not found' });
        return;
      }
      const accountToken = request.headers.authorization?.match(/^Bearer\s+(.+)$/i)?.[1];
      const account = accountToken ? await accountForToken(accountUrl, accountToken) : undefined;
      if (!account?.entitlements.generation) {
        sendJson(response, 403, { error: 'Managed services unavailable' });
        return;
      }
      const body = await requestJson(request);
      const workspaceId = typeof body['workspaceId'] === 'string' ? body['workspaceId'] : '';
      const model = typeof body['model'] === 'string' ? body['model'] : '';
      if (
        !/^[A-Za-z0-9._:-]{1,200}$/.test(workspaceId) ||
        !/^[A-Za-z0-9._:/-]{1,200}$/.test(model) ||
        body['provider'] !== 'openrouter' ||
        body['workload'] !== 'coding'
      ) {
        sendJson(response, 400, { error: 'Invalid inference grant request' });
        return;
      }
      const token = `vgai_local_${randomBytes(32).toString('base64url')}`;
      const grant: ManagedInferenceGrant = {
        kind: 'inference-grant',
        audience: 'vgai-generative-gateway',
        userId: account.user.id,
        workspaceId,
        provider: 'openrouter',
        workload: 'coding',
        model,
        maxRequests: 64,
        maxCredits: 100,
        jti: randomBytes(18).toString('base64url'),
        expiresAt: Date.now() + 15 * 60_000,
      };
      grants.set(token, grant);
      sendJson(response, 200, { grant: token });
    } catch (error) {
      sendJson(response, 500, { error: error instanceof Error ? error.message : String(error) });
    }
  });
  await new Promise<void>((resolve, reject) => {
    server.once('error', reject);
    server.listen(0, '127.0.0.1', resolve);
  });
  const address = server.address() as AddressInfo;
  return {
    url: `http://127.0.0.1:${address.port}`,
    close: () =>
      new Promise<void>((resolve, reject) =>
        server.close((error) => (error ? reject(error) : resolve())),
      ),
  };
}

async function sealedFalFetch(
  twinUrl: string,
  input: RequestInfo | URL,
  init?: RequestInit,
): Promise<Response> {
  const original = new Request(input, init);
  const target = new URL(original.url);
  if (
    target.hostname !== 'fal.run' &&
    target.hostname !== 'queue.fal.run' &&
    target.hostname !== 'rest.fal.ai' &&
    target.hostname !== 'api.fal.ai' &&
    !target.hostname.endsWith('.fal.run')
  ) {
    throw new Error(`The sealed managed-services world blocked external upstream ${target.host}.`);
  }
  const headers = new Headers(original.headers);
  headers.set('x-fal-target-url', original.url);
  const body =
    original.method === 'GET' || original.method === 'HEAD'
      ? undefined
      : new Uint8Array(await original.arrayBuffer());
  return fetch(new URL(`${target.pathname}${target.search}`, twinUrl), {
    method: original.method,
    headers,
    ...(body && body.byteLength > 0 ? { body } : {}),
  });
}

function requiredLoopbackUrl(name: 'CLERK_TWIN_URL' | 'POLAR_TWIN_URL' | 'FAL_TWIN_URL'): string {
  const value = process.env[name];
  if (!value) throw new Error(`${name} is required. Run through dev:managed-services.`);
  const url = new URL(value);
  if (url.protocol !== 'http:' || !['127.0.0.1', 'localhost', '[::1]'].includes(url.hostname)) {
    throw new Error(`${name} must be an HTTP loopback Twin URL.`);
  }
  return url.toString().replace(/\/$/, '');
}

async function startWorldWatchdog(): Promise<void> {
  const instancePath = process.env['VOLTER_WORLD_INSTANCE'];
  if (!instancePath || process.env['VOLTER_WORLD_NAME'] !== 'vgai-managed-services-dev') return;
  const instance = JSON.parse(await readFile(instancePath, 'utf8')) as { createdAt?: unknown };
  if (typeof instance.createdAt !== 'string') {
    throw new Error('The managed-services development world has no creation identity.');
  }
  const watchdog = spawn(
    process.execPath,
    [
      resolve('scripts/managed-services-dev-watchdog.mjs'),
      String(process.pid),
      instancePath,
      instance.createdAt,
    ],
    { cwd: process.cwd(), detached: true, stdio: 'ignore', env: process.env },
  );
  watchdog.unref();
}
