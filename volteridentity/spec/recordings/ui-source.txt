type Result = { data?: unknown; error?: { message?: string; error_description?: string } | null };

function el<K extends keyof HTMLElementTagNameMap>(tag: K, props: Partial<HTMLElementTagNameMap[K]> = {}, ...children: (Node | string)[]): HTMLElementTagNameMap[K] {
  const node = Object.assign(document.createElement(tag), props);
  for (const child of children) node.append(child);
  return node;
}

function icon(name: keyof typeof ICONS): SVGElement {
  const holder = document.createElement("span");
  holder.innerHTML = ICONS[name];
  return holder.firstElementChild as SVGElement;
}

function render(title: string, lede: Node | string | undefined, ...children: Node[]): void {
  renderFor(undefined, title, lede, ...children);
}

function renderFor(product: Product | undefined, title: string, lede: Node | string | undefined, ...children: Node[]): void {
  app.replaceChildren(...productRow(product), el("h1", {}, title), ...(lede === undefined ? [] : [el("p", { className: "lede" }, lede)]), ...children);
  document.title = `${title} · Volter`;
}

function note(): HTMLParagraphElement {
  return el("p", { className: "note", role: "status" });
}

function say(target: HTMLElement, text: string, kind: "error" | "info" | "ok" = "info"): void {
  target.textContent = text;
  target.className = `note ${kind}`;
}

/** Follows a redirect Better Auth answers with, and reports an error in the page. */
function settle(result: Result, target: HTMLElement): boolean {
  if (result.error) { say(target, result.error.message ?? result.error.error_description ?? "Something went wrong.", "error"); return false; }
  const data = result.data as { url?: string; redirect?: boolean; uri?: string } | undefined;
  const next = data?.url ?? data?.uri;
  if (next) { location.href = next; return true; }
  return false;
}

/** Runs a button's action with the button showing it is working, and leaves it pressable again if nothing navigated. */
async function busy(button: HTMLButtonElement, working: string, action: () => Promise<boolean | void>): Promise<void> {
  const label = [...button.childNodes];
  button.disabled = true;
  button.lastChild!.textContent = working;
  let navigating = false;
  try { navigating = (await action()) === true; }
  finally {
    if (!navigating) { button.replaceChildren(...label); button.disabled = false; }
  }
}

function labelled(text: string, iconName?: keyof typeof ICONS): (Node | string)[] {
  return [...(iconName ? [icon(iconName)] : []), el("span", {}, text)];
}

interface Product { name: string; logo?: string }

/** The product that sent the person here, from its registration (name, and its logo when it registered one). The client plugin signs the request with the page's query. */
async function requestingProduct(): Promise<Product | undefined> {
  const clientId = query.get("client_id");
  if (!clientId) return;
  const answer = await client.$fetch<{ client_name?: string; logo_uri?: string }>("/oauth2/public-client-prelogin", { method: "POST", body: { client_id: clientId } }).catch(() => undefined);
  const name = answer?.data?.client_name;
  if (!name) return;
  const logo = answer.data?.logo_uri;
  return { name, ...(logo && /^https:\/\//u.test(logo) ? { logo } : {}) };
}

/** Who is asking, above the heading, when the product registered a logo; without one the lede names it. */
function productRow(product: Product | undefined): Node[] {
  if (!product?.logo) return [];
  return [el("div", { className: "product" },
    el("img", { src: product.logo, alt: "", referrerPolicy: "no-referrer" }),
    el("span", { className: "label" }, product.name))];
}

/** The small mark on the way this browser signed in last. */
function lastUsed(method: string, target: HTMLElement): void {
  if (client.isLastUsedLoginMethod(method)) target.append(el("span", { className: "last-used", textContent: "Last used" }));
}

async function signInPage(): Promise<void> {
  // Where to land after a plain sign-in; an OAuth request continues on its own.
  const callbackURL = query.get("return_to") ?? "/";
  // Someone already signed in, with no product waiting, has nothing to do here.
  if (!query.has("client_id") && (await client.getSession()).data) { location.replace(callbackURL); return; }
  const product = await requestingProduct();
  const lede = product ? el("span", {}, "to continue to ", el("b", {}, product.name)) : "with your Volter account";
  const status = note();

  const providers: Node[] = [];
  for (const provider of ["github", "google"] as const) {
    if (!methods.has(provider)) continue;
    const button = el("button", { type: "button", className: "provider" }, ...labelled(`Continue with ${PROVIDERS[provider]}`, provider));
    lastUsed(provider, button);
    button.addEventListener("click", () => void busy(button, `Opening ${PROVIDERS[provider]}…`, async () => settle(await client.signIn.social({ provider, callbackURL }), status)));
    providers.push(button);
  }

  const parts: Node[] = [status];
  if (providers.length) parts.push(el("div", { className: "stack" }, ...providers));

  if (methods.has("email")) {
    if (providers.length) parts.push(el("div", { className: "or", textContent: "or" }));
    // `webauthn` lets the browser offer a saved passkey right in this field.
    const email = el("input", { type: "email", id: "email", name: "email", placeholder: "you@company.com", autocomplete: "email webauthn", required: true });
    const submit = el("button", { type: "submit", className: "primary" }, el("span", {}, "Continue with email"));
    lastUsed("email-otp", submit);
    const form = el("form", { className: "stack" }, el("div", {}, el("label", { htmlFor: "email", textContent: "Email" }), email), submit);
    form.addEventListener("submit", (event) => {
      event.preventDefault();
      void busy(submit, "Sending a code…", async () => {
        const address = email.value.trim();
        const result = await client.emailOtp.sendVerificationOtp({ email: address, type: "sign-in" });
        if (result.error) { settle(result, status); return; }
        codeStep(address, callbackURL, product);
      });
    });
    parts.push(form);
  }

  if (methods.has("passkey")) {
    const passkey = el("button", { type: "button", className: "link" }, ...labelled("Sign in with a passkey"));
    lastUsed("passkey", passkey);
    passkey.addEventListener("click", () => void busy(passkey, "Waiting for your passkey…", async () => {
      const result = await client.signIn.passkey();
      if (settle(result, status)) return true;
      if (!result.error) { location.href = callbackURL; return true; }
    }));
    parts.push(el("div", { className: "links" }, passkey));
  }

  renderFor(product, "Sign in", lede, ...parts);
  (app.querySelector<HTMLElement>("#email") ?? app.querySelector<HTMLElement>("button"))?.focus();
  // A saved passkey is offered in the email field's autofill (conditional mediation); choosing it signs in.
  if (methods.has("passkey") && await PublicKeyCredential?.isConditionalMediationAvailable?.().catch(() => false)) {
    // It waits silently: an autofill nobody chose, or one the explicit passkey sign-in replaced, is no error.
    void client.signIn.passkey({ autoFill: true }).then((result) => {
      if (result.error) return;
      if (!settle(result, status)) location.href = callbackURL;
    }, () => undefined);
  }
}

/** The second step of an emailed code: its own screen, with a way back and a way to send again. */
function codeStep(address: string, callbackURL: string, product: Product | undefined): void {
  const status = note();
  const code = el("input", { id: "code", className: "code", inputMode: "numeric", autocomplete: "one-time-code", maxLength: 6, pattern: "[0-9]{6}", placeholder: "••••••", required: true });
  const submit = el("button", { type: "submit", className: "primary" }, el("span", {}, product ? `Continue to ${product.name}` : "Sign in"));
  const form = el("form", { className: "stack" }, el("div", {}, el("label", { htmlFor: "code", textContent: "Code" }), code), submit);
  form.addEventListener("submit", (event) => {
    event.preventDefault();
    void busy(submit, "Signing in…", async () => {
      const result = await client.signIn.emailOtp({ email: address, otp: code.value.trim() });
      if (settle(result, status)) return true;
      if (!result.error) { location.href = callbackURL; return true; }
    });
  });
  // Six digits typed or pasted is the whole answer.
  code.addEventListener("input", () => { code.value = code.value.replace(/\D/gu, "").slice(0, 6); if (code.value.length === 6) form.requestSubmit(); });
  const again = el("button", { type: "button", className: "link" }, el("span", {}, "Send a new code"));
  again.addEventListener("click", () => void busy(again, "Sending…", async () => {
    const result = await client.emailOtp.sendVerificationOtp({ email: address, type: "sign-in" });
    if (result.error) settle(result, status); else say(status, `A new code is on its way to ${address}.`, "ok");
  }));
  const back = el("button", { type: "button", className: "link" }, el("span", {}, "Use a different email"));
  back.addEventListener("click", () => void signInPage());
  renderFor(product, "Check your email", el("span", {}, "We sent a 6-digit code to ", el("b", {}, address), ". It expires in 10 minutes."), status, form, el("div", { className: "links" }, back, again));
  code.focus();
}

async function consentPage(): Promise<void> {
  const status = note();
  const scopes = (query.get("scope") ?? "openid").split(" ").filter(Boolean);
  const product = await requestingProduct();
  const name = product?.name ?? query.get("client_id") ?? "An application";
  const words: Record<string, string> = {
    openid: "Know who you are",
    profile: "See your name and picture",
    email: "See your email address",
    offline_access: "Stay signed in until you sign out",
  };
  const allow = el("button", { type: "button", className: "primary" }, el("span", {}, "Allow"));
  const deny = el("button", { type: "button" }, el("span", {}, "Deny"));
  allow.addEventListener("click", () => void busy(allow, "Allowing…", async () => settle(await client.oauth2.consent({ accept: true }), status)));
  deny.addEventListener("click", () => void busy(deny, "Denying…", async () => settle(await client.oauth2.consent({ accept: false }), status)));
  renderFor(product, "Allow access", el("span", {}, el("b", {}, name), " would like to:"),
    status, el("ul", { className: "scopes" }, ...scopes.map((scope) => el("li", {}, words[scope] ?? scope))),
    el("div", { className: "row" }, deny, allow));
}

async function devicePage(): Promise<void> {
  const status = note();
  const session = await client.getSession();
  if (!session.data) { location.href = `/login?return_to=${encodeURIComponent(location.pathname + location.search)}`; return; }
  const input = el("input", { id: "user-code", className: "code", placeholder: "ABCD-EFGH", value: query.get("user_code") ?? "", autocomplete: "off", required: true });
  const approve = el("button", { type: "submit", className: "primary" }, el("span", {}, "Sign in this device"));
  const deny = el("button", { type: "button" }, el("span", {}, "Deny"));
  const form = el("form", { className: "stack" }, el("div", {}, el("label", { htmlFor: "user-code", textContent: "Code" }), input), el("div", { className: "row" }, deny, approve));
  const userCode = (): string => input.value.trim().toUpperCase();
  form.addEventListener("submit", (event) => {
    event.preventDefault();
    void busy(approve, "Signing in…", async () => {
      // The signed-in person claims the code first (GET /device), then approves it.
      const claimed = await client.device({ query: { user_code: userCode() } });
      if (claimed.error) { settle(claimed, status); return; }
      const result = await client.device.approve({ userCode: userCode() });
      if (!settle(result, status) && !result.error) render("Device signed in", "You can go back to the app or terminal that asked.");
    });
  });
  deny.addEventListener("click", () => void busy(deny, "Denying…", async () => {
    const claimed = await client.device({ query: { user_code: userCode() } });
    if (claimed.error) { settle(claimed, status); return; }
    const result = await client.device.deny({ userCode: userCode() });
    if (!settle(result, status) && !result.error) render("Device refused", "Nothing was signed in.");
  }));
  render("Sign in a device", el("span", {}, "Signed in as ", el("b", {}, session.data.user.email), ". Enter the code the app or terminal shows."), status, form);
  input.focus();
}

async function accountPage(): Promise<void> {
  const session = await client.getSession();
  if (!session.data) { location.href = "/login"; return; }
  const user = session.data.user;
  const status = note();
  const accounts = await client.listAccounts();
  const linked = new Set((accounts.data ?? []).map((account) => account.providerId));
  const rows: Node[] = [];
  for (const provider of ["github", "google"] as const) {
    if (!methods.has(provider)) continue;
    if (linked.has(provider)) { rows.push(el("li", {}, icon(provider), el("span", { className: "label" }, PROVIDERS[provider]), el("span", { className: "chip" }, "linked"))); continue; }
    const link = el("button", { type: "button", className: "small" }, el("span", {}, "Link"));
    link.addEventListener("click", () => void busy(link, "Opening…", async () => settle(await client.linkSocial({ provider, callbackURL: "/" }), status)));
    rows.push(el("li", {}, icon(provider), el("span", { className: "label" }, PROVIDERS[provider]), link));
  }
  if (methods.has("email")) rows.push(el("li", {}, icon("email"), el("span", { className: "label" }, "Emailed code"), el("span", { className: "chip" }, "on")));
  const addPasskey = el("button", { type: "button", className: "small" }, el("span", {}, "Add"));
  addPasskey.addEventListener("click", () => void busy(addPasskey, "Waiting…", async () => {
    const result = await client.passkey.addPasskey();
    if (result?.error) settle(result, status); else say(status, "Passkey added. You can sign in with it next time.", "ok");
  }));
  rows.push(el("li", {}, icon("passkey"), el("span", { className: "label" }, "Passkey"), addPasskey));
  const signOut = el("button", { type: "button" }, el("span", {}, "Sign out"));
  signOut.addEventListener("click", () => void busy(signOut, "Signing out…", async () => { await client.signOut(); location.href = "/login"; return true; }));
  // Without a picture of their own, the person gets the brand's placeholder, seeded by their id so it stays theirs.
  const picture = user.image || `https://brand.volter.ai/runner/svg?seed=${encodeURIComponent(user.id)}`;
  const avatar = el("div", { className: "avatar" }, el("img", { src: picture, alt: "" }));
  render("Your account", undefined,
    el("div", { className: "who" }, avatar, el("div", {}, el("div", { className: "name" }, user.name || user.email), el("div", { className: "email" }, user.email))),
    el("h2", {}, "Ways to sign in"), status, el("ul", { className: "methods" }, ...rows),
    el("div", { className: "stack spaced" }, signOut));
}

/** An invitation a product sent through this service (ADR-0002): the invited person, signed in with that address, accepts or declines. */
async function invitationPage(): Promise<void> {
  const id = query.get("id") ?? "";
  const session = await client.getSession();
  if (!session.data) { location.href = `/login?return_to=${encodeURIComponent(location.pathname + location.search)}`; return; }
  const status = note();
  const found = await client.organization.getInvitation({ query: { id } });
  if (found.error || !found.data) {
    render("Invitation not found", el("span", {}, "It may have expired or been withdrawn, or it was sent to another address than ", el("b", {}, session.data.user.email), "."));
    return;
  }
  const invitation = found.data as { organizationName?: string; role?: string; inviterEmail?: string };
  const organization = invitation.organizationName ?? "an organization";
  const accept = el("button", { type: "button", className: "primary" }, el("span", {}, "Accept"));
  const decline = el("button", { type: "button" }, el("span", {}, "Decline"));
  accept.addEventListener("click", () => void busy(accept, "Joining…", async () => {
    const result = await client.organization.acceptInvitation({ invitationId: id });
    if (!settle(result, status) && !result.error) render(`You joined ${organization}`, "Go back to the product that invited you; it shows the organization now.");
  }));
  decline.addEventListener("click", () => void busy(decline, "Declining…", async () => {
    const result = await client.organization.rejectInvitation({ invitationId: id });
    if (!settle(result, status) && !result.error) render("Invitation declined", "Nothing was joined.");
  }));
  render(`Join ${organization}`, el("span", {}, invitation.inviterEmail ? `${invitation.inviterEmail} invited ` : "You're invited as ", el("b", {}, session.data.user.email), invitation.role ? ` as ${invitation.role}.` : "."),
    status, el("div", { className: "row" }, decline, accept));
}

/**
 * The brand's artwork, made by a rule rather than drawn (Plotter Logic): a
 * grid of short strokes turning around a moving centre, a few squares for the
 * agents in it, and a rare orange stroke. Seeded by the day, so it holds still
 * while a person signs in and differs tomorrow. Its colours are the brand's roles, set by
 * identity.css on the strokes, the squares and the orange stroke's class.
 */
function drawArt(): void {
  const svg = document.getElementById("art");
  if (!svg || !svg.parentElement || getComputedStyle(svg.parentElement).display === "none") return;
  const { width, height } = svg.getBoundingClientRect();
  let seed = Math.floor(Date.now() / 86_400_000) * 2654435761 >>> 0;
  const random = (): number => { seed = (seed * 1664525 + 1013904223) >>> 0; return seed / 4294967296; };
  const cx = width * (0.55 + random() * 0.2), cy = height * (0.35 + random() * 0.2), spin = 0.6 + random() * 0.8;
  const step = 18, parts: string[] = [];
  for (let y = step; y < height - step * 2.5; y += step) {
    for (let x = step; x < width - step; x += step) {
      const dx = x - cx, dy = y - cy, distance = Math.hypot(dx, dy);
      if (distance < 46) continue;
      const angle = Math.atan2(dy, dx) + Math.PI / 2 + spin * Math.exp(-distance / 260) + (random() - 0.5) * 0.25;
      const length = 5 + 7 * Math.min(1, 140 / distance);
      const orange = random() < 0.012;
      const x2 = (x + Math.cos(angle) * length).toFixed(1), y2 = (y + Math.sin(angle) * length).toFixed(1);
      parts.push(`<line x1="${x}" y1="${y}" x2="${x2}" y2="${y2}"${orange ? ' class="hot"' : ""} stroke-width="${orange ? 1.4 : 0.8}" stroke-linecap="square"/>`);
      if (random() < 0.018) parts.push(`<rect x="${x - 3}" y="${y - 3}" width="6" height="6"/>`);
    }
  }
  parts.push(`<line x1="${cx - 30}" y1="${cy}" x2="${cx + 30}" y2="${cy}" stroke-width=".6" stroke-dasharray="3 3"/><line x1="${cx}" y1="${cy - 30}" x2="${cx}" y2="${cy + 30}" stroke-width=".6" stroke-dasharray="3 3"/>`);
  svg.setAttribute("viewBox", `0 0 ${width} ${height}`);
  svg.innerHTML = parts.join("");
}
drawArt();
